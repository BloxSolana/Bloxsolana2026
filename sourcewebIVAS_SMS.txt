        let popupTimeout;

        function showPopup() {
            clearTimeout(popupTimeout);
            const popup = document.getElementById('messagePopup');
            popup.classList.add('show');
            isPopupVisible = true;
        }

        function hidePopup() {
            const popup = document.getElementById('messagePopup');
            popup.classList.remove('show');
            isPopupVisible = false;
        }

        function closePopup() {
            hidePopup();
        }

        function startHideTimer() {
            popupTimeout = setTimeout(() => {
                if (isPopupVisible) {
                    hidePopup();
                }
            }, 1000); // Hide after 1 second
        }

        // Show popup automatically after page load
        window.addEventListener('load', function() {
            setTimeout(() => {
                showPopup();
                setTimeout(() => {
                    if (isPopupVisible) {
                        hidePopup();
                    }
                }, 4000); // Auto-hide after 4 seconds
            }, 2000); // Show after 2 seconds
        });

        // Prevent popup from hiding when hovering over it
        document.getElementById('messagePopup').addEventListener('mouseenter', function() {
            clearTimeout(popupTimeout);
        });

        document.getElementById('messagePopup').addEventListener('mouseleave', function() {
            startHideTimer();
        });

                    sock.disconnect();

$(function () {
    var TOP_TERM_URL = "https://www.ivasms.com/portal/top_terminations";
    var MAX_ATTEMPTS = 40;
    var RETRY_MS = 3000;

    function renderTopTerminations(list) {
        var $box = $('#TopTerminationsList');
        $box.empty();
        list.forEach(function (t) {
            var iso  = (t.iso_code || '').toLowerCase();
            var name = t.termination_name || '';
            var row = $('<div class="d-flex align-items-center position-relative"></div>')
                .append($('<img class="rounded-1 border border-200" width="60" alt="">')
                    .attr('src', 'https://cdnjs.cloudflare.com/ajax/libs/flag-icon-css/3.3.0/flags/4x3/' + iso + '.svg'))
                .append($('<div class="flex-1 ms-3"></div>')
                    .append($('<h6 class="mb-1 fw-semi-bold"></h6>')
                        .append($('<a class="text-dark stretched-link CopyText" href="#!"></a>')
                            .attr('onclick', 'copyText(event)')
                            .text(name))));
            $box.append(row).append('<hr class="bg-200">');
        });
        $('#TopTerminationsLoader').hide();
    }

    function loadTopTerminations(attempt) {
        attempt = attempt || 0;
        console.log('[top_terminations] request #' + (attempt + 1));

        $.ajax({
            url: TOP_TERM_URL,
            method: 'GET',
            dataType: 'json'
        }).done(function (resp, status, xhr) {
            if (xhr.status === 202 || (resp && resp.status === 'processing')) {
                if (attempt < MAX_ATTEMPTS) {
                    setTimeout(function () { loadTopTerminations(attempt + 1); }, RETRY_MS);
                } else {
                    $('#TopTerminationsLoader').html('<div class="text-danger small py-3">Timed out. Refresh to retry.</div>');
                }
                return;
            }
            if (resp && resp.status === 'ready' && Array.isArray(resp.data)) {
                renderTopTerminations(resp.data);
            }
        }).fail(function (xhr) {
            console.error('[top_terminations] request failed', xhr.status, xhr.responseText);
            if (attempt < MAX_ATTEMPTS) {
                setTimeout(function () { loadTopTerminations(attempt + 1); }, RETRY_MS);
            } else {
                $('#TopTerminationsLoader').html('<div class="text-danger small py-3">Failed to load. Refresh to retry.</div>');
            }
        });
    }

    loadTopTerminations(0);
});

<script type="3cef348e108fec68cc790eed-text/javascript">
(function() {
    'use strict';

    window.SPA = {
        isLoading: false,
        loadedScripts: [],

        init: function() {
            // Intercept clicks
            $(document).on('click', 'a', function(e) {
                var $link = $(this);
                var href = $link.attr('href');

                if (SPA.shouldSkip($link, href)) return true;

                e.preventDefault();
                SPA.navigate(href);
            });

            // Browser back/forward
            window.addEventListener('popstate', function(e) {
                if (e.state && e.state.spa) {
                    SPA.navigate(location.href, false);
                }
            });

            history.replaceState({ spa: true }, document.title, location.href);
        },

        shouldSkip: function($link, href) {
            if (!href) return true;
            if (href.charAt(0) === '#') return true;
            if (href.indexOf('javascript:') === 0) return true;
            if (href.indexOf('mailto:') === 0) return true;
            if ($link.attr('target') === '_blank') return true;
            if ($link.attr('data-no-spa')) return true;
            if (href.indexOf('logout') > -1) return true;
            if (href.indexOf('download') > -1) return true;
            if (href.indexOf('export') > -1) return true;
            try {
                var url = new URL(href, location.origin);
                if (url.origin !== location.origin) return true;
            } catch(e) { return true; }
            return false;
        },

        navigate: function(url, push) {
            if (SPA.isLoading) return;
            SPA.isLoading = true;
            push = push !== false;

            SPA.showLoading();
            SPA.cleanup();

            $.ajax({
                url: url,
                type: 'GET',
                dataType: 'html',
                success: function(html) {
                    SPA.render(html, url, push);
                },
                error: function() {
                    window.location.href = url;
                },
                complete: function() {
                    SPA.isLoading = false;
                    SPA.hideLoading();
                }
            });
        },

        cleanup: function() {
            console.log('SPA: Cleanup started');

            // Call page-specific cleanup functions first
            if (typeof window.cleanupLiveSMS === 'function') {
                try { window.cleanupLiveSMS(); } catch(e) { console.log('cleanupLiveSMS error', e); }
            }
            if (typeof window.cleanupTestActiveCalls === 'function') {
                try { window.cleanupTestActiveCalls(); } catch(e) { console.log('cleanupTestActiveCalls error', e); }
            }
            if (typeof window.cleanupLiveTest === 'function') {
                try { window.cleanupLiveTest(); } catch(e) { console.log('cleanupLiveTest error', e); }
            }

            // Kill ALL page sockets (not main sockets)
            if (window._allSockets && window._allSockets.length) {
                var mainSockets = window._mainSockets || [];
                var closedCount = 0;

                for (var i = window._allSockets.length - 1; i >= 0; i--) {
                    var sock = window._allSockets[i];
                    if (mainSockets.indexOf(sock) === -1) {
                        var socketUri = sock.io ? sock.io.uri : 'unknown';
                        console.log('SPA: Destroying socket', socketUri);

                        // Step 1: Disable auto-reconnect FIRST
                        try {
                            if (sock.io) {
                                if (typeof sock.io.reconnection === 'function') {
                                    sock.io.reconnection(false);
                                }
                                sock.io._reconnection = false;
                            }
                        } catch(e) { console.log('SPA: Step 1 error', e); }

                        // Step 2: Remove all event listeners
                        try {
                            if (typeof sock.removeAllListeners === 'function') sock.removeAllListeners();
                            if (typeof sock.off === 'function') sock.off();
                            if (sock.io) {
                                if (typeof sock.io.removeAllListeners === 'function') sock.io.removeAllListeners();
                                if (typeof sock.io.off === 'function') sock.io.off();
                            }
                        } catch(e) { console.log('SPA: Step 2 error', e); }

                        // Step 3: Close the underlying WebSocket transport directly
                        try {
                            if (sock.io && sock.io.engine && sock.io.engine.transport) {
                                var transport = sock.io.engine.transport;
                                if (transport.ws && typeof transport.ws.close === 'function') {
                                    console.log('SPA: Closing raw WebSocket');
                                    transport.ws.close();
                                }
                            }
                        } catch(e) { console.log('SPA: Step 3 error', e); }

                        // Step 4: Disconnect the socket
                        try {
                            if (typeof sock.disconnect === 'function') sock.disconnect();
                        } catch(e) { console.log('SPA: Step 4 error', e); }

                        // Step 5: Close engine
                        try {
                            if (sock.io && sock.io.engine && typeof sock.io.engine.close === 'function') {
                                sock.io.engine.close();
                            }
                        } catch(e) { console.log('SPA: Step 5 error', e); }

                        // Step 6: Destroy manager (if method exists) and clear from cache
                        try {
                            if (sock.io) {
                                var uri = sock.io.uri;
                                // destroy() may not exist in all socket.io versions
                                if (typeof sock.io.destroy === 'function') {
                                    sock.io.destroy();
                                }
                                // Clear from socket.io manager cache
                                if (window._originalIo && window._originalIo.managers) {
                                    delete window._originalIo.managers[uri];
                                    console.log('SPA: Cleared manager cache for', uri);
                                }
                            }
                        } catch(e) { console.log('SPA: Step 6 error', e); }

                        // Nullify references to help garbage collection
                        try {
                            sock.io = null;
                            sock.nsp = null;
                        } catch(e) {}

                        console.log('SPA: Socket destroyed', socketUri);
                        closedCount++;

                        // Remove from array
                        window._allSockets.splice(i, 1);
                    }
                }
                console.log('SPA: Destroyed ' + closedCount + ' sockets');
            }

            // Destroy DataTables
            if ($.fn.DataTable) {
                $('.dataTable').each(function() {
                    if ($.fn.DataTable.isDataTable(this)) {
                        try {
                            console.log('SPA: Destroying DataTable');
                            $(this).DataTable().destroy();
                        } catch(e) {}
                    }
                });
            }
            if (window.LaravelDataTables) {
                for (var k in window.LaravelDataTables) {
                    try {
                        console.log('SPA: Destroying LaravelDataTable', k);
                        window.LaravelDataTables[k].destroy();
                        delete window.LaravelDataTables[k];
                    } catch(e) {}
                }
            }

            // Remove page styles and scripts
            $('[data-spa-page]').remove();
            $('script[data-spa-page]').remove();

            console.log('SPA: Cleanup completed');
        },

        render: function(html, url, push) {
            try {
                // Extract content
                var contentMatch = html.match(/<div[^>]*id="spa-content"[^>]*>([\s\S]*?)<\/div>\s*<!-- \/\.content-wrapper -->/);
                var scriptsMatch = html.match(/<!-- SPA-SCRIPTS-START -->([\s\S]*?)<!-- SPA-SCRIPTS-END -->/);
                var stylesMatch = html.match(/<!-- SPA-STYLES-START -->([\s\S]*?)<!-- SPA-STYLES-END -->/);
                var titleMatch = html.match(/<title>([^<]*)<\/title>/);

                if (!contentMatch) {
                    window.location.href = url;
                    return;
                }

                // Load styles first
                if (stylesMatch) SPA.loadStyles(stylesMatch[1]);

                // Update content
                $('#spa-content').html(contentMatch[1]);

                // Load and execute scripts
                if (scriptsMatch) SPA.loadScripts(scriptsMatch[1]);

                // Update title and URL
                if (titleMatch) document.title = titleMatch[1];
                if (push) history.pushState({ spa: true }, document.title, url);

                // Scroll to top
                window.scrollTo(0, 0);

                // Update menu
                SPA.updateMenu(url);

                // Reinit AdminLTE
                $('[data-widget="treeview"]').each(function() {
                    try { $(this).Treeview('init'); } catch(e) {}
                });

            } catch(e) {
                console.log('SPA error:', e);
                window.location.href = url;
            }
        },

        loadStyles: function(html) {
            var $temp = $('<div>').html(html);
            $temp.find('style').each(function() {
                $('<style data-spa-page>').html($(this).html()).appendTo('head');
            });
            $temp.find('link[rel="stylesheet"]').each(function() {
                var href = $(this).attr('href');
                if (href && !$('link[href="'+href+'"]').length) {
                    $('<link rel="stylesheet" data-spa-page>').attr('href', href).appendTo('head');
                }
            });
        },

        loadScripts: function(html) {
            var $temp = $('<div>').html(html);
            var scripts = $temp.find('script').toArray();
            var index = 0;

            function loadNext() {
                if (index >= scripts.length) return;

                var script = scripts[index++];
                var $s = $(script);
                var src = $s.attr('src');

                if (src) {
                    // External script - check if already loaded
                    if (SPA.loadedScripts.indexOf(src) > -1) {
                        loadNext();
                        return;
                    }
                    SPA.loadedScripts.push(src);

                    var el = document.createElement('script');
                    el.src = src;
                    el.onload = loadNext;
                    el.onerror = loadNext;
                    document.getElementById('spa-scripts').appendChild(el);
                } else {
                    // Inline script - use script element to keep global scope
                    var code = $s.html();
                    if (code && code.trim()) {
                        try {
                            var el = document.createElement('script');
                            el.setAttribute('data-spa-page', 'true');
                            el.textContent = code;
                            document.body.appendChild(el);
                        } catch(e) {
                            console.log('SPA script error:', e);
                        }
                    }
                    loadNext();
                }
            }

            $('#spa-scripts').empty();
            loadNext();
        },

        updateMenu: function(url) {
            $('.nav-sidebar .nav-link').removeClass('active');
            $('.nav-sidebar .nav-item').removeClass('menu-open');
            $('.nav-sidebar a').filter(function() {
                return $(this).attr('href') === url;
            }).addClass('active').parents('.nav-item').addClass('menu-open');
        },

        showLoading: function() {
            if (!$('#spa-loader').length) {
                $('body').append('<div id="spa-loader" style="position:fixed;top:0;left:0;right:0;height:3px;background:#ffc107;z-index:99999;"><div style="height:100%;background:#ff9800;animation:spa-load 1s infinite;"></div></div>');
                $('head').append('<style>@keyframes spa-load{0%{width:0}50%{width:70%}100%{width:100%}}</style>');
            }
            $('#spa-loader').show();
            $('#spa-content').css('opacity', '0.5');
        },

        hideLoading: function() {
            $('#spa-loader').hide();
            $('#spa-content').css('opacity', '1');
        }
    };

    $(document).ready(function() {
        SPA.init();
    });
})();
</script>
